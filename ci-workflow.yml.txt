# СКОПИРУЙ СОДЕРЖИМОЕ ЭТОГО ФАЙЛА В .github/workflows/build.yml
# (у агента нет workflow-скоупа, он не может закоммитить этот путь)
#
# Создать: https://github.com/kotovviktor542-hub/Maybe/new/main
#   имя файла: .github/workflows/build.yml
#
# И включить: Settings -> Actions -> General ->
#             Workflow permissions = Read and write permissions
# ============================================================

name: Build APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: write

env:
  GODOT_VERSION: 4.3
  EXPORT_NAME: Maybe

jobs:
  build:
    runs-on: ubuntu-latest
    container:
      image: barichello/godot-ci:4.3
    steps:
      - uses: actions/checkout@v4

      - name: Setup export templates
        run: |
          mkdir -p ~/.local/share/godot/export_templates
          mv /root/.local/share/godot/export_templates/${GODOT_VERSION}.stable ~/.local/share/godot/export_templates/

      - name: Setup Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Generate debug keystore
        run: |
          keytool -keyalg RSA -genkeypair -alias androiddebugkey \
            -keypass android -keystore debug.keystore -storepass android \
            -dname "CN=Android Debug,O=Android,C=US" -validity 9999 -deststoretype pkcs12
          echo "GODOT_ANDROID_KEYSTORE_DEBUG=$PWD/debug.keystore" >> $GITHUB_ENV
          echo "GODOT_ANDROID_KEYSTORE_DEBUG_USER=androiddebugkey" >> $GITHUB_ENV
          echo "GODOT_ANDROID_KEYSTORE_DEBUG_PASSWORD=android" >> $GITHUB_ENV

      - name: Import project
        run: godot --headless --import . || true

      - name: Export APK
        run: |
          mkdir -p build/android
          godot --headless --export-debug "Android" build/android/$EXPORT_NAME.apk

      - name: Upload artifact
        uses: actions/upload-artifact@v4
        with:
          name: apk
          path: build/android/*.apk

      - name: Create release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v0.1.0-build${{ github.run_number }}
          files: build/android/*.apk
          generate_release_notes: true
